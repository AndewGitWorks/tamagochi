func databaseURLFromEnv() (string, error) {
        if dsn := os.Getenv("DATABASE_URL"); dsn != "" {
                return dsn, nil
        }
        user, password, dbName := os.Getenv("POSTGRES_USER"), os.Getenv("POSTGRES_PASSWORD"), os.Getenv("POSTGRES_DB")
        if user == "" || password == "" || dbName == "" {
                return "", fmt.Errorf("set DATABASE_URL or POSTGRES_USER, POSTGRES_PASSWORD, and POSTGRES_DB")
        }
        host := envOrDefault("POSTGRES_HOST", "localhost")
        port := envOrDefault("POSTGRES_PORT", "5432")
        sslMode := envOrDefault("POSTGRES_SSLMODE", "disable")
        u := url.URL{Scheme: "postgres", User: url.UserPassword(user, password), Host: host + ":" + port, Path: "/" + dbName}
        q := u.Query()
        q.Set("sslmode", sslMode)
        u.RawQuery = q.Encode()
        return u.String(), nil
}

func envOrDefault(name, fallback string) string {
        if value := os.Getenv(name); value != "" {
                return value
        }
        return fallback
}

func envDurationSeconds(name string, fallback time.Duration) time.Duration {
        value := os.Getenv(name)
        if value == "" {
                return fallback
        }
        seconds, err := strconv.Atoi(value)
        if err != nil || seconds <= 0 {
                log.Fatalf("%s must be a positive number of seconds", name)
        }
        return time.Duration(seconds) * time.Second
}

// loadProjectEnv reads the root .env when the API is started from either the
// repository root or backend/. Existing shell environment values take priority.
func loadProjectEnv() error {
        err := loadEnvFile(".env")
        if errors.Is(err, os.ErrNotExist) {
                err = loadEnvFile("../.env")
        }
        return err
}

func loadEnvFile(path string) error {
        file, err := os.Open(path)
        if err != nil {
                return err
        }
        defer file.Close()
        scanner := bufio.NewScanner(file)
        lineNumber := 0
        for scanner.Scan() {
                lineNumber++
                line := strings.TrimSpace(scanner.Text())
                if line == "" || strings.HasPrefix(line, "#") {
                        continue
                }
                key, value, found := strings.Cut(line, "=")
                if !found || strings.TrimSpace(key) == "" {
                        return fmt.Errorf("invalid %s line %d", path, lineNumber)
                }
                if _, alreadySet := os.LookupEnv(key); !alreadySet {
                        if err := os.Setenv(key, strings.TrimSpace(value)); err != nil {
                                return fmt.Errorf("set %s from %s: %w", key, path, err)
                        }
                }
        }
        return scanner.Err()
}

func (h *AuthHandler) Register(w http.ResponseWriter, r *http.Request) {

        }

        writeJSON(w, http.StatusOK, map[string]string{
                "status": "ok",
        })
}
func (h *AuthHandler) Refresh(w http.ResponseWriter, r *http.Request) {
        token, ok := bearerToken(r.Header.Get("Authorization"))
        if !ok {
                writeAuthError(w, security.ErrInvalidToken)
                return
        }
        pair, err := h.auth.Refresh(r.Context(), token)
        if err != nil {
                writeAuthError(w, err)
                return
        }
        writeJSON(w, http.StatusOK, pair)
}

func decodeJSON(w http.ResponseWriter, r *http.Request, target any) bool {
        r.Body = http.MaxBytesReader(w, r.Body, maxRequestBodySize)
        decoder := json.NewDecoder(r.Body)
        decoder.DisallowUnknownFields()
        if err := decoder.Decode(target); err != nil {
                writeJSON(w, http.StatusBadRequest, map[string]string{"error": "invalid_request", "message": "request body must be valid JSON"})
                return false
        }
        if err := decoder.Decode(new(any)); err != io.EOF {
                writeJSON(w, http.StatusBadRequest, map[string]string{"error": "invalid_request", "message": "request body must contain one JSON value"})
                return false
        }
        return true
}

func bearerToken(value string) (string, bool) {
        parts := strings.Fields(value)
        returnToken := len(parts) == 2 && strings.EqualFold(parts[0], "Bearer") && parts[1] != ""
        if !returnToken {
                return "", false
        }
        return parts[1], true
}

func writeAuthError(w http.ResponseWriter, err error) {
        switch {
        case errors.Is(err, users.ErrValidation):
                writeJSON(w, http.StatusBadRequest, map[string]string{"error": "invalid_request", "message": err.Error()})
        case errors.Is(err, users.ErrLoginTaken), errors.Is(err, users.ErrUserConflict):
                writeJSON(w, http.StatusConflict, map[string]string{"error": "login_taken", "message": err.Error()})
        case errors.Is(err, users.ErrInvalidCredentials), errors.Is(err, security.ErrInvalidToken):
                w.Header().Set("WWW-Authenticate", `Bearer realm="tamagochi-api"`)
                writeJSON(w, http.StatusUnauthorized, map[string]string{"error": "authentication_error", "message": err.Error()})
        default:
                writeJSON(w, http.StatusInternalServerError, map[string]string{"error": "internal_error", "message": "an unexpected error occurred"})
        }
}

func writeJSON(w http.ResponseWriter, status int, value any) {
        w.Header().Set("Content-Type", "application/json; charset=utf-8")
        w.WriteHeader(status)
        _ = json.NewEncoder(w).Encode(value)
}

type userContextKey struct{}

func currentUser(r *http.Request) users.User {
        return r.Context().Value(userContextKey{}).(users.User)
}

// RequireUser validates an access token and passes the current user to an endpoint.
// Route composition is kept in the routes package; handlers only implement endpoints.
func RequireUser(auth *users.AuthService, admin bool, next http.HandlerFunc) http.HandlerFunc {
        return func(w http.ResponseWriter, r *http.Request) {
                token, ok := bearerToken(r.Header.Get("Authorization"))
                if !ok {
                        writeJSON(w, http.StatusUnauthorized, map[string]string{"error": "access token required"})
                        return
                }
                user, err := auth.AuthenticateAccess(r.Context(), token)
                if err != nil {
                        writeAuthError(w, err)
                        return
                }
                if admin && user.Role != "ADMIN" {
                        writeJSON(w, http.StatusForbidden, map[string]string{"error": "admin role required"})
                        return
                }
                next(w, r.WithContext(context.WithValue(r.Context(), userContextKey{}, user)))
        }
}

func pathID(w http.ResponseWriter, r *http.Request, key string) (int64, bool) {
        id, err := strconv.ParseInt(r.PathValue(key), 10, 64)
        if err != nil || id <= 0 {
                writeCRUDError(w, domain.ErrValidation)
                return 0, false
        }
        return id, true
}

func pagination(w http.ResponseWriter, r *http.Request) (int, int, bool) {
        limit, offset := 20, 0
        var err error
        if value := r.URL.Query().Get("limit"); value != "" {
                limit, err = strconv.Atoi(value)
                if err != nil || limit < 1 || limit > 100 {
                        writeCRUDError(w, domain.ErrValidation)
                        return 0, 0, false
                }
        }
        if value := r.URL.Query().Get("offset"); value != "" {
                offset, err = strconv.Atoi(value)
                if err != nil || offset < 0 {
                        writeCRUDError(w, domain.ErrValidation)
                        return 0, 0, false
                }
        }
        return limit, offset, true
}

func writeCRUDError(w http.ResponseWriter, err error) {
        status, message := http.StatusInternalServerError, "an unexpected error occurred"
        switch {
        case errors.Is(err, domain.ErrNotFound), errors.Is(err, users.ErrUserNotFound):
                status, message = http.StatusNotFound, "entity was not found"
        case errors.Is(err, domain.ErrValidation), errors.Is(err, users.ErrValidation):
                status, message = http.StatusBadRequest, "invalid entity data"
        case errors.Is(err, domain.ErrConflict), errors.Is(err, users.ErrUserConflict), errors.Is(err, users.ErrLoginTaken):
                status, message = http.StatusConflict, "entity conflicts with existing data"
        case errors.Is(err, domain.ErrOrderLocked):
                status, message = http.StatusConflict, "only pending orders can be edited"
        case errors.Is(err, domain.ErrOrderTransition):
                status, message = http.StatusConflict, "order status transition is not allowed"
        }
        writeJSON(w, status, map[string]string{"error": message})
}

func respond(w http.ResponseWriter, status int, value any, err error) {
        if err != nil {
                writeCRUDError(w, err)
                return
        }
        writeJSON(w, status, value)
}

func respondFound(w http.ResponseWriter, value any, found bool, err error) {
        if err != nil {
                writeCRUDError(w, err)
                return
        }
        if !found {
                writeCRUDError(w, domain.ErrNotFound)
                return
        }
        writeJSON(w, http.StatusOK, value)
}

func respondDeleted(w http.ResponseWriter, err error) {
        if err != nil {
                writeCRUDError(w, err)
                return
        }
        w.WriteHeader(http.StatusNoContent)
}

func cors(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                log.Printf("CORS: %s %s", r.Method, r.URL.Path)
                w.Header().Set("Access-Control-Allow-Origin", "*")
                w.Header().Set("Access-Control-Allow-Methods", "GET, POST, PUT, DELETE, OPTIONS")
                w.Header().Set("Access-Control-Allow-Headers", "Content-Type, Authorization")
                if r.Method == http.MethodOptions {
                        w.WriteHeader(http.StatusNoContent)
                        return
                }
                next.ServeHTTP(w, r)
        })
}

// Actor is the minimum identity required to apply order access rules. The HTTP
// layer converts the authenticated user to this type before calling the service.
type Actor struct {

        Role string
}

func (a Actor) IsAdmin() bool { return a.Role == "ADMIN" }

// Service is the business boundary for carts and orders. Repositories persist
// data and protect transactions; this service owns permissions and state rules.
type Service struct {
        orders Store
        items  ItemStore
        number func() (string, error)
}

// Cart is a pending order together with its current line items. A user has at
// most one cart; the database enforces that invariant with a partial index.
type Cart struct {
        Order Order  `json:"order"`
        Items []Item `json:"items"`
}

func NewService(orders Store, items ItemStore) *Service {
        return &Service{orders: orders, items: items, number: newOrderNumber}
}

func (s *Service) Create(ctx context.Context, userID int64) (Order, error) {
        if userID <= 0 {
                return Order{}, domain.ErrValidation
        }
        number, err := s.number()
        if err != nil {
                return Order{}, err
        }
        return s.orders.Create(ctx, Order{UserID: userID, Number: number, Status: "PENDING"})
}

// OpenCart returns the user's active cart or creates it. If two concurrent
// requests try to create it, the unique database index lets one succeed and
// the other request reads the newly created cart.
func (s *Service) OpenCart(ctx context.Context, userID int64) (Order, error) {
        return s.pendingCart(ctx, userID, true)
}

func (s *Service) GetCart(ctx context.Context, userID int64) (Cart, error) {
        if userID <= 0 {
                return Cart{}, domain.ErrValidation
        }
        order, err := s.pendingCart(ctx, userID, false)
        if err != nil {
                return Cart{}, err
        }
        items, err := s.items.ListByOrder(ctx, order.ID, 100, 0)
        if err != nil {
                return Cart{}, err
        }
        return Cart{Order: order, Items: items}, nil
}

// AddToCart increments an existing line instead of creating duplicate lines
// for the same product. The repository recalculates total_price atomically.
func (s *Service) AddToCart(ctx context.Context, userID, productID, quantity int64) (Cart, error) {
        if userID <= 0 || productID <= 0 || quantity <= 0 {
                return Cart{}, domain.ErrValidation
        }
        order, err := s.pendingCart(ctx, userID, true)
        if err != nil {
                return Cart{}, err
        }
        item, found, err := s.items.FindByProduct(ctx, order.ID, productID)
        if err != nil {
                return Cart{}, err
        }
        if found {
                if quantity > (1<<63-1)-item.Quantity {
                        return Cart{}, domain.ErrValidation
                }
                if _, err = s.items.Update(ctx, Item{ID: item.ID, OrderID: order.ID, Quantity: item.Quantity + quantity}); err != nil {
                        return Cart{}, err
                }
        } else if _, err = s.items.Create(ctx, Item{OrderID: order.ID, ProductID: productID, Quantity: quantity}); err != nil {
                return Cart{}, err
        }
        return s.GetCart(ctx, userID)
}

func (s *Service) UpdateCartItem(ctx context.Context, userID, itemID, quantity int64) (Cart, error) {
        if userID <= 0 || itemID <= 0 || quantity <= 0 {
                return Cart{}, domain.ErrValidation
        }
        order, err := s.pendingCart(ctx, userID, false)
        if err != nil {
                return Cart{}, err
        }
        if _, err = s.items.Update(ctx, Item{ID: itemID, OrderID: order.ID, Quantity: quantity}); err != nil {
                return Cart{}, err
        }
        return s.GetCart(ctx, userID)
}

func (s *Service) RemoveFromCart(ctx context.Context, userID, itemID int64) (Cart, error) {
        if userID <= 0 || itemID <= 0 {
                return Cart{}, domain.ErrValidation
        }
        order, err := s.pendingCart(ctx, userID, false)
        if err != nil {
                return Cart{}, err
        }
        if err = s.items.Delete(ctx, itemID, order.ID); err != nil {
                return Cart{}, err
        }
        return s.GetCart(ctx, userID)
}

// Checkout freezes the cart and records the delivery address with the order.
// The address is copied here so future profile changes never alter its history.
func (s *Service) Checkout(ctx context.Context, userID int64, deliveryAddress string) (Order, error) {
        deliveryAddress = strings.TrimSpace(deliveryAddress)
        if userID <= 0 || utf8.RuneCountInString(deliveryAddress) < 5 || utf8.RuneCountInString(deliveryAddress) > 500 {
                return Order{}, domain.ErrValidation
        }
        order, err := s.pendingCart(ctx, userID, false)
        if err != nil {
                return Order{}, err
        }
        items, err := s.items.ListByOrder(ctx, order.ID, 1, 0)
        if err != nil {
                return Order{}, err
        }
        if len(items) == 0 {
                return Order{}, domain.ErrValidation
        }
        return s.orders.Checkout(ctx, order.ID, userID, deliveryAddress)
}

// Pay is a deterministic payment simulator for the university project. A
// production integration would call a provider and apply its signed callback.
func (s *Service) Pay(ctx context.Context, actor Actor, orderID int64, approved bool) (Order, error) {
        order, err := s.owned(ctx, actor, orderID)
        if err != nil {
                return Order{}, err
        }
        if order.Status != "AWAITING_PAYMENT" && order.Status != "PAYMENT_FAILED" {
                return Order{}, domain.ErrOrderTransition
        }
        if approved {
                order.Status = "PAID"
        } else {
                order.Status = "PAYMENT_FAILED"
        }
        return s.orders.Update(ctx, order)
}

        return s.owned(ctx, actor, orderID)
}

func (s *Service) List(ctx context.Context, userID int64, limit, offset int) ([]Order, error) {
        if userID <= 0 {
                return nil, domain.ErrValidation
        }
        return s.orders.ListByUser(ctx, userID, limit, offset)

func (s *Service) ChangeStatus(ctx context.Context, actor Actor, orderID int64, target string) (Order, error) {
        order, err := s.owned(ctx, actor, orderID)
        if err != nil {
                return Order{}, err
        }
        if !actor.IsAdmin() {
                if target != "CANCELLED" {
                        return Order{}, domain.ErrOrderTransition
                }
                return s.orders.Cancel(ctx, order.ID, actor.ID)
        }
        if !allowedTransition(order.Status, target) {
                return Order{}, domain.ErrOrderTransition
        }
        order.Status = target
        return s.orders.Update(ctx, order)
}

func (s *Service) Delete(ctx context.Context, actor Actor, orderID int64) error {
        order, err := s.owned(ctx, actor, orderID)
        if err != nil {
                return err
        }
        if order.Status != "PENDING" && order.Status != "CANCELLED" {
                return domain.ErrOrderLocked
        }
        return s.orders.Delete(ctx, order.ID)
}

        }

                return Item{}, err
        }
        return s.items.Update(ctx, Item{ID: itemID, OrderID: orderID, Quantity: quantity})
}

func (s *Service) DeleteItem(ctx context.Context, actor Actor, orderID, itemID int64) error {
        if _, err := s.owned(ctx, actor, orderID); err != nil {
                return err
        }
        return s.items.Delete(ctx, itemID, orderID)
}

func (s *Service) owned(ctx context.Context, actor Actor, orderID int64) (Order, error) {
        if actor.ID <= 0 || orderID <= 0 {
                return Order{}, domain.ErrValidation
        }
        order, found, err := s.orders.FindByID(ctx, orderID)
        if err != nil {
                return Order{}, err
        }
        if !found || (!actor.IsAdmin() && order.UserID != actor.ID) {
                // Do not disclose whether an order belonging to another user exists.
                return Order{}, domain.ErrNotFound
        }
        return order, nil
}

func (s *Service) pendingCart(ctx context.Context, userID int64, create bool) (Order, error) {
        if userID <= 0 {
                return Order{}, domain.ErrValidation
        }
        cart, found, err := s.orders.FindPendingByUser(ctx, userID)
        if err != nil {
                return Order{}, err
        }
        if found {
                return cart, nil
        }
        if !create {
                return Order{}, domain.ErrNotFound
        }
        cart, err = s.Create(ctx, userID)
        if err == nil {
                return cart, nil
        }
        if !errors.Is(err, domain.ErrConflict) {
                return Order{}, err
        }
        cart, found, findErr := s.orders.FindPendingByUser(ctx, userID)
        if findErr != nil {
                return Order{}, findErr
        }
        if !found {
                return Order{}, err
        }
        return cart, nil
}

func allowedTransition(from, to string) bool {
        switch from {
        case "PENDING":
                return to == "CANCELLED"
        case "AWAITING_PAYMENT", "PAYMENT_FAILED":
                return to == "CANCELLED"
        case "PAID":
                return to == "DELIVERING"
        case "DELIVERING":
                return to == "DELIVERED"
        }
        return false
}

func newOrderNumber() (string, error) {
        var value [16]byte
        if _, err := rand.Read(value[:]); err != nil {
                return "", err
        }
        return hex.EncodeToString(value[:]), nil
}

// Keep errors.Is useful for callers that want to identify terminal-state
// failures without leaking persistence details.
func IsTransitionError(err error) bool { return errors.Is(err, domain.ErrOrderTransition) }

func NewAuthService(config AuthConfig) *AuthService {
        if config.AccessTokenTTL <= 0 {
                config.AccessTokenTTL = 15 * time.Minute
        }
        if config.RefreshTokenTTL <= 0 {
                config.RefreshTokenTTL = 30 * 24 * time.Hour
        }
        if config.RefreshSessions == nil {
                config.RefreshSessions = NewMemoryRefreshSessionStore()
        }
        return &AuthService{store: config.Store, jwtSecret: config.JWTSecret, issuer: config.Issuer,
                accessTTL: config.AccessTokenTTL, refreshTTL: config.RefreshTokenTTL,

func (s *AuthService) Register(ctx context.Context, input RegisterInput) (TokenPair, error) {
        if err := validateRegistration(input); err != nil {
                return TokenPair{}, err
        }
        hash, err := security.HashPassword(input.Password)
        if err != nil {
                return TokenPair{}, err
        }
        user, err := s.store.Create(ctx, User{Login: input.Login, Name: input.Name, Surname: input.Surname,
                Email: input.Email, PhoneNumber: input.PhoneNumber, Role: "USER", PasswordHash: hash})
        if err != nil {
                return TokenPair{}, err
        }
        return s.issuePair(ctx, user)
}

func validateRegistration(input RegisterInput) error {
        if !loginPattern.MatchString(input.Login) || len([]rune(input.Name)) < 3 || len([]rune(input.Name)) > 100 ||
                len([]rune(input.Surname)) < 3 || len([]rune(input.Surname)) > 100 || len(input.Email) < 3 || len(input.Email) > 100 || len(input.Password) < 8 ||
                len(input.PhoneNumber) < 10 || len(input.PhoneNumber) > 15 || !phonePattern.MatchString(input.PhoneNumber) {
                return fmt.Errorf("%w: login, name, surname, email, phone number or password does not meet requirements", ErrValidation)
        }
        address, err := mail.ParseAddress(input.Email)
        if err != nil || address.Address != input.Email {
                return fmt.Errorf("%w: email is invalid", ErrValidation)
        }
        return nil
}

func (s *AuthService) Login(ctx context.Context, input LoginInput) (TokenPair, error) {
        user, found, err := s.store.FindByLogin(ctx, input.Login)
        if err != nil {
                return TokenPair{}, err
        }
        if !found || !security.VerifyPassword(input.Password, user.PasswordHash) {
                return TokenPair{}, ErrInvalidCredentials
        }
        return s.issuePair(ctx, user)
}

func (s *AuthService) Refresh(ctx context.Context, refreshToken string) (TokenPair, error) {
        claims, err := security.VerifyJWT(refreshToken, s.jwtSecret, s.issuer, "refresh", s.now())
        if err != nil {
                return TokenPair{}, err
        }
        id, parseErr := strconv.ParseInt(claims.Subject, 10, 64)
        if parseErr != nil {
                return TokenPair{}, security.ErrInvalidToken
        }
        consumed, err := s.refreshSessions.Consume(ctx, claims.ID, id)
        if err != nil {
                return TokenPair{}, err
        }
        if !consumed {
                return TokenPair{}, security.ErrInvalidToken
        }
        user, found, err := s.store.FindByID(ctx, id)
        if err != nil || !found || strconv.FormatInt(user.ID, 10) != claims.Subject || user.Login != claims.Login {
                return TokenPair{}, security.ErrInvalidToken
        }
        return s.issuePair(ctx, user)
}

func (s *AuthService) issuePair(ctx context.Context, user User) (TokenPair, error) {
        now := s.now().UTC()
        access, accessExpiry, _, err := s.issueToken(user, "access", s.accessTTL, now)
        if err != nil {
                return TokenPair{}, err
        }
        refresh, refreshExpiry, refreshID, err := s.issueToken(user, "refresh", s.refreshTTL, now)
        if err != nil {
                return TokenPair{}, err
        }
        if err := s.refreshSessions.Create(ctx, RefreshSession{ID: refreshID, UserID: user.ID, ExpiresAt: refreshExpiry}); err != nil {
                return TokenPair{}, err
        }
        return TokenPair{AccessToken: access, RefreshToken: refresh, TokenType: "Bearer", AccessExpiresAt: accessExpiry.Format(time.RFC3339), RefreshExpiresAt: refreshExpiry.Format(time.RFC3339)}, nil
}

func (s *AuthService) issueToken(user User, tokenType string, ttl time.Duration, now time.Time) (token string, expiresAt time.Time, id string, err error) {
        id, err = randomID()
        if err != nil {
                return "", time.Time{}, "", err
        }
        expiresAt = now.Add(ttl)
        claims := security.Claims{Issuer: s.issuer, Subject: strconv.FormatInt(user.ID, 10), Login: user.Login, Name: user.Name, Role: user.Role,
                TimeCreation: now.Format(time.RFC3339), TimeExpiration: expiresAt.Format(time.RFC3339),
                IssuedAt: now.Unix(), ExpiresAt: expiresAt.Unix(), ID: id, TokenType: tokenType}
        token, err = security.SignJWT(claims, s.jwtSecret)
        return token, expiresAt, id, err
}

func randomID() (string, error) {
        bytes := make([]byte, 16)
        if _, err := rand.Read(bytes); err != nil {
                return "", err
        }
        return base64.RawURLEncoding.EncodeToString(bytes), nil
}

// AuthenticateAccess checks the token and reads the current role from the store.
// Deleted users and users whose login changed cannot keep using an old token.
func (s *AuthService) AuthenticateAccess(ctx context.Context, token string) (User, error) {
        claims, err := security.VerifyJWT(token, s.jwtSecret, s.issuer, "access", s.now())
        if err != nil {
                return User{}, err
        }
        id, err := strconv.ParseInt(claims.Subject, 10, 64)
        if err != nil || id <= 0 {
                return User{}, security.ErrInvalidToken
        }
        user, found, err := s.store.FindByID(ctx, id)
        if err != nil {
                return User{}, err
        }
        if !found || user.Login != claims.Login {
                return User{}, security.ErrInvalidToken
        }
        return user, nil
}

// Store is the persistence boundary for user CRUD. The PostgreSQL implementation
// also creates the initial profile atomically; MemoryStore is only for unit tests.
type Store interface {

// RefreshSessionStore persists single-use refresh-token identifiers. It is
// separate from Store so that a database-backed implementation can rotate a
// token atomically without coupling it to user CRUD.
type RefreshSessionStore interface {
        Create(ctx context.Context, session RefreshSession) error
        Consume(ctx context.Context, id string, userID int64) (bool, error)
}

type RefreshSession struct {
        ID        string `json:"id"`
        UserID    int64  `json:"user_id"`
        ExpiresAt time.Time
}

// MemoryStore is used by isolated service tests and local development.
type MemoryStore struct {
        mu      sync.RWMutex
        nextID  int64
        byID    map[int64]User
        byLogin map[string]int64
        byEmail map[string]int64
}

func NewMemoryStore() *MemoryStore {
        return &MemoryStore{byID: make(map[int64]User), byLogin: make(map[string]int64), byEmail: make(map[string]int64)}
}

func (s *MemoryStore) Create(_ context.Context, user User) (User, error) {
        s.mu.Lock()
        defer s.mu.Unlock()
        if _, exists := s.byLogin[user.Login]; exists {
                return User{}, ErrLoginTaken
        }
        if _, exists := s.byEmail[user.Email]; exists {
                return User{}, ErrUserConflict
        }
        s.nextID++
        user.ID, user.Version = s.nextID, 1
        if user.Role == "" {
                user.Role = "USER"
        }
        s.byID[user.ID], s.byLogin[user.Login], s.byEmail[user.Email] = user, user.ID, user.ID
        return user, nil
}

func (s *MemoryStore) FindByLogin(_ context.Context, login string) (User, bool, error) {
        s.mu.RLock()
        defer s.mu.RUnlock()
        id, ok := s.byLogin[login]
        if !ok {
                return User{}, false, nil
        }
        return s.byID[id], true, nil
}

func (s *MemoryStore) FindByID(_ context.Context, id int64) (User, bool, error) {
        s.mu.RLock()
        defer s.mu.RUnlock()
        user, ok := s.byID[id]
        return user, ok, nil
}

func (s *MemoryStore) Update(_ context.Context, user User) (User, error) {
        s.mu.Lock()
        defer s.mu.Unlock()
        old, ok := s.byID[user.ID]
        if !ok {
                return User{}, ErrUserNotFound
        }
        if id, exists := s.byLogin[user.Login]; exists && id != user.ID {
                return User{}, ErrLoginTaken
        }
        if id, exists := s.byEmail[user.Email]; exists && id != user.ID {
                return User{}, ErrUserConflict
        }
        delete(s.byLogin, old.Login)
        delete(s.byEmail, old.Email)
        user.Version = old.Version + 1
        if user.Role == "" {
                user.Role = old.Role
        }
        s.byID[user.ID], s.byLogin[user.Login], s.byEmail[user.Email] = user, user.ID, user.ID
        return user, nil
}

func (s *MemoryStore) Delete(_ context.Context, id int64) error {
        s.mu.Lock()
        defer s.mu.Unlock()
        user, ok := s.byID[id]
        if !ok {
                return ErrUserNotFound
        }
        delete(s.byID, id)
        delete(s.byLogin, user.Login)
        delete(s.byEmail, user.Email)
        return nil
}

// MemoryRefreshSessionStore is only the default for isolated unit tests.
// The running API uses repository.PostgresRefreshSessionStore.
type MemoryRefreshSessionStore struct {
        mu       sync.Mutex
        sessions map[string]RefreshSession
}

func NewMemoryRefreshSessionStore() *MemoryRefreshSessionStore {
        return &MemoryRefreshSessionStore{sessions: make(map[string]RefreshSession)}
}

func (s *MemoryRefreshSessionStore) Create(_ context.Context, session RefreshSession) error {
        s.mu.Lock()
        defer s.mu.Unlock()
        s.sessions[session.ID] = session
        return nil
}

func (s *MemoryRefreshSessionStore) Consume(_ context.Context, id string, userID int64) (bool, error) {
        s.mu.Lock()
        defer s.mu.Unlock()
        session, found := s.sessions[id]
        if !found || session.UserID != userID || !session.ExpiresAt.After(time.Now()) {
                return false, nil
        }
        delete(s.sessions, id)
        return true, nil
}

import (

var ErrInvalidToken = errors.New("bearer token is invalid or has expired")

type Claims struct {
        Issuer         string `json:"iss"`
        Subject        string `json:"sub"`
        Login          string `json:"login"`
        Name           string `json:"name"`
        Role           string `json:"role"`
        TimeExpiration string `json:"time_expiration"`
        TimeCreation   string `json:"time_creation"`
        IssuedAt       int64  `json:"iat"`
        ExpiresAt      int64  `json:"exp"`
        ID             string `json:"jti"`
        TokenType      string `json:"token_type"`
}

func SignJWT(claims Claims, secret []byte) (string, error) {
        header, err := json.Marshal(struct {
                Algorithm string `json:"alg"`
                Type      string `json:"typ"`
        }{Algorithm: "HS256", Type: "JWT"})
        if err != nil {
                return "", err
        }
        payload, err := json.Marshal(claims)
        if err != nil {
                return "", err
        }
        unsigned := base64.RawURLEncoding.EncodeToString(header) + "." + base64.RawURLEncoding.EncodeToString(payload)
        return unsigned + "." + signature(unsigned, secret), nil
}

func VerifyJWT(token string, secret []byte, issuer, expectedType string, now time.Time) (Claims, error) {
        parts := strings.Split(token, ".")
        if len(parts) != 3 {
                return Claims{}, ErrInvalidToken
        }
        if subtle.ConstantTimeCompare([]byte(parts[2]), []byte(signature(parts[0]+"."+parts[1], secret))) != 1 {
                return Claims{}, ErrInvalidToken
        }

        var header struct {
                Algorithm string `json:"alg"`
                Type      string `json:"typ"`
        }
        headerJSON, err := base64.RawURLEncoding.DecodeString(parts[0])
        if err != nil || json.Unmarshal(headerJSON, &header) != nil || header.Algorithm != "HS256" || header.Type != "JWT" {
                return Claims{}, ErrInvalidToken
        }
        payloadJSON, err := base64.RawURLEncoding.DecodeString(parts[1])
        if err != nil {
                return Claims{}, ErrInvalidToken
        }
        var claims Claims
        if json.Unmarshal(payloadJSON, &claims) != nil ||
                claims.Issuer != issuer || claims.TokenType != expectedType ||
                claims.Subject == "" || claims.Login == "" || claims.Name == "" || claims.Role == "" || claims.ID == "" ||
                claims.IssuedAt == 0 || claims.ExpiresAt <= now.Unix() {
                return Claims{}, ErrInvalidToken
        }
        return claims, nil
}

func signature(unsigned string, secret []byte) string {
        mac := hmac.New(sha256.New, secret)
        mac.Write([]byte(unsigned))
        return base64.RawURLEncoding.EncodeToString(mac.Sum(nil))
}

import (

        "crypto/sha256"

        "strings"

const passwordIterations = 210_000
const passwordKeyLength = 24 // 192-bit key; keeps the complete encoding within VARCHAR(64).
const passwordSaltLength = 16
const passwordHashVersion = 1
const passwordIterationBytes = 3

// HashPassword uses PBKDF2-HMAC-SHA256 with a unique random salt. The compact
// versioned encoding stores version, iteration count, salt, and derived key in
// one base64 value so it fits the existing password_hashed VARCHAR(64) column.
func HashPassword(password string) (string, error) {
        salt := make([]byte, passwordSaltLength)
        if _, err := rand.Read(salt); err != nil {
                return "", fmt.Errorf("create password salt: %w", err)
        }
        derived := pbkdf2SHA256([]byte(password), salt, passwordIterations, passwordKeyLength)
        encoded := make([]byte, 1+passwordIterationBytes+len(salt)+len(derived))
        encoded[0] = passwordHashVersion
        encoded[1] = byte(passwordIterations >> 16)
        encoded[2] = byte(passwordIterations >> 8 & 0xff)
        encoded[3] = byte(passwordIterations & 0xff)
        copy(encoded[4:], salt)
        copy(encoded[4+len(salt):], derived)
        return base64.RawURLEncoding.EncodeToString(encoded), nil
}

func VerifyPassword(password, encoded string) bool {
        // Keep support for hashes created by the previous textual representation.
        if strings.HasPrefix(encoded, "pbkdf2-sha256$") {
                return verifyLegacyPassword(password, encoded)
        }
        data, err := base64.RawURLEncoding.DecodeString(encoded)
        if err != nil || len(data) != 1+passwordIterationBytes+passwordSaltLength+passwordKeyLength || data[0] != passwordHashVersion {
                return false
        }
        iterations := int(data[1])<<16 | int(data[2])<<8 | int(data[3])
        if iterations < 1 {
                return false
        }
        salt := data[4 : 4+passwordSaltLength]
        expected := data[4+passwordSaltLength:]
        actual := pbkdf2SHA256([]byte(password), salt, iterations, len(expected))
        return subtle.ConstantTimeCompare(expected, actual) == 1
}

func verifyLegacyPassword(password, encoded string) bool {
        parts := strings.Split(encoded, "$")
        if len(parts) != 4 || parts[0] != "pbkdf2-sha256" {
                return false
        }
        iterations, err := strconv.Atoi(parts[1])
        if err != nil || iterations < 1 {
                return false
        }
        salt, err := base64.RawURLEncoding.DecodeString(parts[2])
        if err != nil {
                return false
        }
        expected, err := base64.RawURLEncoding.DecodeString(parts[3])
        if err != nil || len(expected) == 0 {
                return false
        }
        actual := pbkdf2SHA256([]byte(password), salt, iterations, len(expected))
        return subtle.ConstantTimeCompare(expected, actual) == 1
}

func pbkdf2SHA256(password, salt []byte, iterations, length int) []byte {
        result := make([]byte, 0, length)
        for block := uint32(1); len(result) < length; block++ {
                mac := hmac.New(sha256.New, password)
                mac.Write(salt)
                mac.Write([]byte{byte(block >> 24), byte(block >> 16), byte(block >> 8), byte(block)})
                u := mac.Sum(nil)
                t := append([]byte(nil), u...)
                for round := 1; round < iterations; round++ {
                        mac = hmac.New(sha256.New, password)
                        mac.Write(u)
                        u = mac.Sum(nil)
                        for i := range t {
                                t[i] ^= u[i]
                        }
                }
                result = append(result, t...)
        }
        return result[:length]
}

func mapCRUDError(err error) error {
        if err == nil {
                return nil
        }
        if errors.Is(err, sql.ErrNoRows) {
                return domain.ErrNotFound
        }
        var state sqlStateError
        if errors.As(err, &state) {
                switch state.SQLState() {
                case "23505", "23503", "23001":
                        return fmt.Errorf("%w: %v", domain.ErrConflict, err)
                case "23514", "23502", "22003", "22P02", "22001":
                        return fmt.Errorf("%w: %v", domain.ErrValidation, err)
                }
        }
        return err
}

func requireAffected(result sql.Result, err error) error {
        if err != nil {
                return mapCRUDError(err)
        }
        count, err := result.RowsAffected()
        if err != nil {
                return err
        }
        if count == 0 {
                return domain.ErrNotFound
        }
        return nil
}

        }
        return limit
}

        err = tx.QueryRowContext(ctx, `UPDATE tamagochi.order_items SET quantity=$1 WHERE id=$2 AND order_id=$3 RETURNING `+itemColumns,

        if err = requireAffected(result, err); err != nil {
                return err
        }
        if err = updateOrderTotal(ctx, tx, orderID); err != nil {
                return err
        }
        return mapCRUDError(tx.Commit())
}

func lockPendingOrder(ctx context.Context, tx *sql.Tx, id int64) error {
        var status string
        err := tx.QueryRowContext(ctx, `SELECT status FROM tamagochi.orders WHERE id=$1 FOR UPDATE`, id).Scan(&status)
        if err != nil {
                return mapCRUDError(err)
        }
        if status != "PENDING" {
                return domain.ErrOrderLocked
        }
        return nil
}

func updateOrderTotal(ctx context.Context, tx *sql.Tx, orderID int64) error {
        _, err := tx.ExecContext(ctx, `UPDATE tamagochi.orders SET total_price=
 (SELECT COALESCE(SUM(quantity * price), 0) FROM tamagochi.order_items WHERE order_id=$1) WHERE id=$1`, orderID)
        return mapCRUDError(err)
}

// PostgresRefreshSessionStore keeps only refresh-token IDs. The JWT itself is
// never stored, so a database leak cannot be used as a bearer credential.
type PostgresRefreshSessionStore struct{ db *sql.DB }

func NewPostgresRefreshSessionStore(db *sql.DB) *PostgresRefreshSessionStore {
        return &PostgresRefreshSessionStore{db: db}
}

func (s *PostgresRefreshSessionStore) Create(ctx context.Context, session users.RefreshSession) error {
        _, err := s.db.ExecContext(ctx, `INSERT INTO tamagochi.refresh_sessions (id, user_id, expires_at)
                VALUES ($1, $2, $3)`, session.ID, session.UserID, session.ExpiresAt)
        if err != nil {
                return fmt.Errorf("save refresh session: %w", err)
        }
        return nil
}

// Consume deletes a non-expired session in one statement. Only one concurrent
// refresh request can receive a successful result for the same token ID.
func (s *PostgresRefreshSessionStore) Consume(ctx context.Context, id string, userID int64) (bool, error) {
        result, err := s.db.ExecContext(ctx, `DELETE FROM tamagochi.refresh_sessions
                WHERE id = $1 AND user_id = $2 AND expires_at > CURRENT_TIMESTAMP`, id, userID)
        if err != nil {
                return false, fmt.Errorf("consume refresh session: %w", err)
        }
        affected, err := result.RowsAffected()
        if err != nil {
                return false, fmt.Errorf("read refresh session result: %w", err)
        }
        return affected == 1, nil
}

type PostgresUserStore struct{ db *sql.DB }

func NewPostgresUserStore(db *sql.DB) *PostgresUserStore { return &PostgresUserStore{db: db} }

const userColumns = `id, name, surname, login, email, password_hashed, phone_number, version, role`

func (s *PostgresUserStore) Create(ctx context.Context, user users.User) (users.User, error) {
        tx, err := s.db.BeginTx(ctx, nil)
        if err != nil {
                return users.User{}, fmt.Errorf("begin user registration: %w", err)
        }
        defer func() { _ = tx.Rollback() }()
        if user.Role == "" {
                user.Role = "USER"
        }
        if user.Version == 0 {
                user.Version = 1
        }
        err = tx.QueryRowContext(ctx, `INSERT INTO tamagochi.users
                (name, surname, login, email, password_hashed, phone_number, version, role)
                VALUES ($1,$2,$3,$4,$5,$6,$7,$8) RETURNING id`,
                user.Name, user.Surname, user.Login, user.Email, user.PasswordHash, user.PhoneNumber, user.Version, user.Role).Scan(&user.ID)
        if err != nil {
                return users.User{}, mapConflict(fmt.Errorf("insert user: %w", err))
        }
        if _, err = tx.ExecContext(ctx, `INSERT INTO tamagochi.profiles (user_id) VALUES ($1)`, user.ID); err != nil {
                return users.User{}, fmt.Errorf("create initial user profile: %w", err)
        }
        if err = tx.Commit(); err != nil {
                return users.User{}, fmt.Errorf("commit user registration: %w", err)
        }
        return user, nil
}

func (s *PostgresUserStore) FindByLogin(ctx context.Context, login string) (users.User, bool, error) {
        return s.find(ctx, `SELECT `+userColumns+` FROM tamagochi.users WHERE login=$1`, login)
}

func (s *PostgresUserStore) FindByID(ctx context.Context, id int64) (users.User, bool, error) {
        return s.find(ctx, `SELECT `+userColumns+` FROM tamagochi.users WHERE id=$1`, id)
}

func (s *PostgresUserStore) find(ctx context.Context, query string, arg any) (users.User, bool, error) {
        var user users.User
        err := s.db.QueryRowContext(ctx, query, arg).Scan(&user.ID, &user.Name, &user.Surname, &user.Login, &user.Email,
                &user.PasswordHash, &user.PhoneNumber, &user.Version, &user.Role)
        if errors.Is(err, sql.ErrNoRows) {
                return users.User{}, false, nil
        }
        if err != nil {
                return users.User{}, false, fmt.Errorf("read user: %w", err)
        }
        return user, true, nil
}

func (s *PostgresUserStore) Update(ctx context.Context, user users.User) (users.User, error) {
        if user.Role == "" {
                user.Role = "USER"
        }
        row := s.db.QueryRowContext(ctx, `UPDATE tamagochi.users SET name=$1,surname=$2,login=$3,email=$4,
                password_hashed=$5,phone_number=$6,role=$7,version=version+1 WHERE id=$8
                RETURNING id,name,surname,login,email,password_hashed,phone_number,version,role`,
                user.Name, user.Surname, user.Login, user.Email, user.PasswordHash, user.PhoneNumber, user.Role, user.ID)
        var updated users.User
        err := row.Scan(&updated.ID, &updated.Name, &updated.Surname, &updated.Login, &updated.Email, &updated.PasswordHash, &updated.PhoneNumber, &updated.Version, &updated.Role)
        if errors.Is(err, sql.ErrNoRows) {
                return users.User{}, users.ErrUserNotFound
        }
        if err != nil {
                return users.User{}, mapConflict(fmt.Errorf("update user: %w", err))
        }
        return updated, nil
}

func (s *PostgresUserStore) Delete(ctx context.Context, id int64) error {
        result, err := s.db.ExecContext(ctx, `DELETE FROM tamagochi.users WHERE id=$1`, id)
        if err != nil {
                return mapCRUDError(fmt.Errorf("delete user: %w", err))
        }
        count, err := result.RowsAffected()
        if err != nil {
                return fmt.Errorf("read delete result: %w", err)
        }
        if count == 0 {
                return users.ErrUserNotFound
        }
        return nil
}

type sqlStateError interface{ SQLState() string }

func mapConflict(err error) error {
        var stateErr sqlStateError
        if errors.As(err, &stateErr) && stateErr.SQLState() == "23505" {
                return fmt.Errorf("%w: %v", users.ErrUserConflict, err)
        }
        return err
}
